=== Integration Tests

[source,java]
----
@SpringBootTest
class ChatQueryServiceIntegrationTest {
    @Autowired
    private ChatQueryService queryService;

    @Test
    void testQueryProcessing() {
        ToolContext context = ToolContext.builder()
            .put("tenantId", "test-tenant")
            .build();
        
        QueryResult result = queryService.processQuery(
            "Show active users", context);
        
        assertNotNull(result);
    }
}
